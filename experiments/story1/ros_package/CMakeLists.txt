cmake_minimum_required(VERSION 3.0.2)
project(course_story1)
add_compile_options(-std=c++14 -O2)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs quadrotor_msgs nav_msgs visualization_msgs rviz pluginlib sensor_msgs pcl_conversions)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS})
add_executable(trajectory_server src/trajectory_server.cpp)
add_dependencies(trajectory_server ${catkin_EXPORTED_TARGETS})
target_link_libraries(trajectory_server ${catkin_LIBRARIES})
add_executable(display_relay src/display_relay.cpp)
add_dependencies(display_relay ${catkin_EXPORTED_TARGETS})
target_link_libraries(display_relay ${catkin_LIBRARIES})
add_library(course_render_stats SHARED src/render_stats.cpp)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
target_link_libraries(course_render_stats ${catkin_LIBRARIES} Qt5::Widgets)
find_package(PCL REQUIRED COMPONENTS common kdtree)
add_executable(fast_trajectory_server src/fast_trajectory_server.cpp)
target_include_directories(fast_trajectory_server PRIVATE ${PCL_INCLUDE_DIRS})
add_dependencies(fast_trajectory_server ${catkin_EXPORTED_TARGETS})
target_link_libraries(fast_trajectory_server ${catkin_LIBRARIES} ${PCL_LIBRARIES})
